export const LaneSwitcher = ({ lanes }) => {
  const [activeLane, setActiveLane] = useState(lanes[0].id)
  const [activeItem, setActiveItem] = useState(lanes[0].items[0]?.href ?? null)

  const lane = lanes.find((l) => l.id === activeLane)
  const current = lane?.items.find((item) => item.href === activeItem) ?? lane?.items[0]

  return (
    <div className="not-prose space-y-6">
      <div className="flex flex-wrap gap-2">
        {lanes.map((l) => (
          <button
            key={l.id}
            onClick={() => {
              setActiveLane(l.id)
              setActiveItem(l.items[0]?.href ?? null)
            }}
            className={`px-4 py-2 text-sm rounded-full font-medium transition-colors ${
              activeLane === l.id
                ? 'bg-[#0A0B0D] text-white dark:bg-white dark:text-[#0A0B0D]'
                : 'bg-gray-100 dark:bg-white/10 text-black dark:text-white'
            }`}
          >
            {l.label}
          </button>
        ))}
      </div>
      <p className="text-sm text-gray-600 dark:text-gray-400">{lane.tagline}</p>
      <div className="flex flex-col lg:flex-row gap-6 bg-gray-100 dark:bg-[#141414] rounded-2xl">
        <div className="flex flex-col w-full lg:w-1/2 space-y-2 p-4">
          {lane.items.map((item) => (
            <a
              key={item.href}
              href={item.href}
              onMouseOver={() => setActiveItem(item.href)}
              className={`flex items-start gap-1.5 text-left px-5 py-3 rounded-2xl transition-all ${
                current?.href === item.href ? 'bg-white dark:bg-white/5' : ''
              }`}
            >
              <div className="flex-shrink-0 w-10 h-10 flex items-center justify-center rounded-lg border border-gray-200 dark:border-white/10 mr-3 bg-background-light dark:bg-background-dark">
                <Icon icon={item.icon} size="16" />
              </div>
              <div className="flex-1">
                <h3 className="font-medium text-base text-black dark:text-white">{item.title}</h3>
                <p className="text-sm text-gray-600 dark:text-gray-400">{item.description}</p>
              </div>
              <div
                className={`flex items-center justify-center min-w-[24px] self-center ${
                  current?.href === item.href ? 'opacity-100' : 'opacity-0'
                }`}
              >
                <div className="hidden dark:block">
                  <Icon icon="chevron-right" size="14" color="#FFF" />
                </div>
                <div className="block dark:hidden">
                  <Icon icon="chevron-right" size="14" color="#0A0B0D" />
                </div>
              </div>
            </a>
          ))}
        </div>
        <div className="w-full h-fit lg:w-1/2 rounded-2xl flex items-center justify-center overflow-hidden min-h-[320px] p-4">
          {lane.image ? (
            <img
              src={lane.image}
              alt={lane.label}
              className="w-full h-full object-cover rounded-xl"
            />
          ) : null}
        </div>
      </div>
    </div>
  )
}
