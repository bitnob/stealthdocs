## GET request example

These examples show how to make an authenticated **GET** request to the Bitnob API using the signing function defined above. For GET requests without a body, pass null as the body parameter. The signature is computed over just your client ID, timestamp, and nonce.

```javascript
const { generateAuthHeaders } = require('./auth');

const CLIENT_ID = process.env.CLIENT_ID;
const CLIENT_SECRET = process.env.CLIENT_SECRET;
const BASE_URL = 'https://api.bitnob.com';

async function getBalances() {
  const authHeaders = generateAuthHeaders(CLIENT_ID, CLIENT_SECRET, null);

  const response = await fetch(`${BASE_URL}/api/balances`, {
    method: 'GET',
    headers: {
      'Content-Type': 'application/json',
      ...authHeaders,
    },
  });

  return response.json();
}

getBalances()
  .then((data) => console.log('Balances:', data))
  .catch((err) => console.error(err));
```

```go
package main

import (
	"encoding/json"
	"fmt"
	"net/http"
	"os"

	"yourproject/auth"
)

var (
	ClientID     = os.Getenv("CLIENT_ID")
	ClientSecret = os.Getenv("CLIENT_SECRET")
	BaseURL      = "https://api.bitnob.com"
)

func getBalances() (map[string]interface{}, error) {
	headers, err := auth.GenerateAuthHeaders(ClientID, ClientSecret, nil)
	if err != nil {
		return nil, err
	}

	req, _ := http.NewRequest("GET", BaseURL+"/api/balances", nil)
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("X-Auth-Client", headers.XAuthClient)
	req.Header.Set("X-Auth-Timestamp", headers.XAuthTimestamp)
	req.Header.Set("X-Auth-Nonce", headers.XAuthNonce)
	req.Header.Set("X-Auth-Signature", headers.XAuthSignature)

	client := &http.Client{}
	resp, err := client.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	var result map[string]interface{}
	json.NewDecoder(resp.Body).Decode(&result)
	return result, nil
}

func main() {
	balances, err := getBalances()
	if err != nil {
		fmt.Println("Error:", err)
		return
	}
	fmt.Printf("Balances: %v\n", balances)
}
```

## POST request example

For **POST**/**PUT** requests, pass the JSON body to generateAuthHeaders so it is included in the HMAC computation. Any change to the payload, even reordering keys, will invalidate the signature, so sign the exact byte sequence you send on the wire.

<Note>
  Use the /api/whoami endpoint to quickly verify your authentication is set up correctly before making other API calls.
</Note>

```python
import os
import requests
from auth import generate_auth_headers

CLIENT_ID = os.getenv('CLIENT_ID')
CLIENT_SECRET = os.getenv('CLIENT_SECRET')
BASE_URL = 'https://api.bitnob.com'

def get_balances():
    headers = generate_auth_headers(CLIENT_ID, CLIENT_SECRET)
    resp = requests.get(
        f'{BASE_URL}/api/balances',
        headers={'Content-Type': 'application/json', **headers}
    )
    return resp.json()

if __name__ == '__main__':
    print('Balances:', get_balances())
```

## Validate authentication

Use the /api/whoami endpoint to verify your credentials are working correctly. This endpoint accepts any HTTP method and any payload, and returns your authenticated client info.

It works with HMAC signature authentication.

### Response fields

1. successboolean

   Whether the request was successfully authenticated. Returns true when the HMAC signature is valid.

2. messagestring

   Human-readable status message. 'Authenticated' when the HMAC signature is valid.

3. dataobject

   The authenticated client info. Contains the fields below.

4. data.activeboolean

   Whether the client is currently active. Inactive clients cannot make authenticated requests.

5. data.active_company_idstring

   UUID of the company this client is currently acting on behalf of.

6. data.auth_methodstring

   The authentication mode used to sign the request. Always 'hmac'.

7. data.client_idstring

   UUID of the client (API app) that made the request.

8. data.client_namestring

   Human-readable name of the client, as configured on the Bitnob dashboard.

9. data.environmentstring

   The environment the request was made against. One of 'sandbox' or 'live'.

10. data.permissionsarray \| null

   List of scoped permissions granted to this client. Null when the client has no scoped permissions (e.g. full account-level access).

11. timestampstring

RFC 3339 / ISO 8601 server timestamp of when the request was processed (UTC).

```javascript
const { generateAuthHeaders } = require('./auth');

const CLIENT_ID = process.env.CLIENT_ID;
const CLIENT_SECRET = process.env.CLIENT_SECRET;
const BASE_URL = 'https://api.bitnob.com';

async function createCustomer(customerData) {
  const body = JSON.stringify(customerData);
  const authHeaders = generateAuthHeaders(CLIENT_ID, CLIENT_SECRET, body);

  const response = await fetch(`${BASE_URL}/api/customers`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...authHeaders,
    },
    body,
  });

  return response.json();
}

createCustomer({
  email: 'john.doe@example.com',
  first_name: 'John',
  last_name: 'Doe',
  phone_number: '+1234567890',
  customer_type: 'individual',
})
  .then((data) => console.log('Customer created:', data))
  .catch((err) => console.error(err));
```
