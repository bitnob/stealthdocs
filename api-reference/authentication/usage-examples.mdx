These examples use the `generateAuthHeaders` (Node.js) and `generate_auth_headers` (Python) helpers from [Authentication](/api-reference/authentication/authentication#reference-implementations). Save the helper as `auth.js` or `auth.py` next to your code.

The one rule that matters: **sign exactly the bytes you send.** For a GET, sign an empty body. For a POST, pass the same body to the helper and to your HTTP client.

## GET request

<CodeGroup>

```javascript Node.js
const { generateAuthHeaders } = require('./auth');

const CLIENT_ID = process.env.CLIENT_ID;
const CLIENT_SECRET = process.env.CLIENT_SECRET;
const BASE_URL = 'https://api.bitnob.com';

async function getBalances() {
  // GET requests have no body, so sign an empty payload
  const headers = generateAuthHeaders(CLIENT_ID, CLIENT_SECRET, null);
  const response = await fetch(`${BASE_URL}/api/balances`, { headers });
  return response.json();
}

getBalances().then((data) => console.log(JSON.stringify(data, null, 2)));
```

```python Python
import os
import requests
from auth import generate_auth_headers

CLIENT_ID = os.environ["CLIENT_ID"]
CLIENT_SECRET = os.environ["CLIENT_SECRET"]
BASE_URL = "https://api.bitnob.com"

# GET requests have no body, so sign an empty payload
headers = generate_auth_headers(CLIENT_ID, CLIENT_SECRET)
response = requests.get(f"{BASE_URL}/api/balances", headers=headers)
print(response.json())
```

</CodeGroup>

## POST request

<CodeGroup>

```javascript Node.js
const { generateAuthHeaders } = require('./auth');

const CLIENT_ID = process.env.CLIENT_ID;
const CLIENT_SECRET = process.env.CLIENT_SECRET;
const BASE_URL = 'https://api.bitnob.com';

async function validateAddress(address, chain) {
  const body = { address, chain };
  // Sign the same body you send: the helper and fetch both use JSON.stringify
  const headers = generateAuthHeaders(CLIENT_ID, CLIENT_SECRET, body);
  const response = await fetch(`${BASE_URL}/api/addresses/validate`, {
    method: 'POST',
    headers: { ...headers, 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });
  return response.json();
}

validateAddress('TRMPscTMrXg21YuDFuvpXRUkwEbipHEZXN', 'tron').then((data) =>
  console.log(JSON.stringify(data, null, 2))
);
```

```python Python
import os
import requests
from auth import generate_auth_headers

CLIENT_ID = os.environ["CLIENT_ID"]
CLIENT_SECRET = os.environ["CLIENT_SECRET"]
BASE_URL = "https://api.bitnob.com"

body = {"address": "TRMPscTMrXg21YuDFuvpXRUkwEbipHEZXN", "chain": "tron"}

# requests serializes json= the same way the helper's json.dumps does,
# so the bytes sent match the bytes signed
headers = generate_auth_headers(CLIENT_ID, CLIENT_SECRET, body)
response = requests.post(f"{BASE_URL}/api/addresses/validate", json=body, headers=headers)
print(response.json())
```

</CodeGroup>

```json Response
{
  "success": true,
  "message": "Address validation completed",
  "data": {
    "valid": true,
    "address": "TRMPscTMrXg21YuDFuvpXRUkwEbipHEZXN",
    "chain": "tron"
  },
  "timestamp": "2026-10-18T07:32:46.692379293Z"
}
```

If you build the body string yourself, for example compact JSON, pass that exact string to the helper and send that same string. A body that differs from the signed one by even a space fails with `401` `Invalid HMAC signature`.

## Confirm your credentials

[`GET /api/whoami`](/api-reference/identity/who-am-i) returns the client your signature authenticated as. It's the quickest way to check signing works, and to confirm which environment your secret selects.

```json
{
  "success": true,
  "message": "Authenticated",
  "data": {
    "active": true,
    "active_company_id": "b1ebb43e-068f-4b6a-bbab-5c3027bc1510",
    "auth_method": "hmac",
    "client_id": "9213f0c7-486f-4c8a-9c57-ab3fe68ee696",
    "client_name": "DC",
    "environment": "sandbox",
    "permissions": null
  },
  "timestamp": "2026-10-18T07:33:30.881661173Z"
}
```

| Field | Meaning |
|---|---|
| `data.environment` | `sandbox` or `live`, set by the secret you signed with |
| `data.active_company_id` | The company the client acts for |
| `data.auth_method` | Always `hmac` |
| `data.permissions` | The client's scoped permissions, or `null` when it has full account access |
